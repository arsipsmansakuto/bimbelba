function onOpen() { SpreadsheetApp.getUi().createMenu('🚀 SIM Presensi') .addItem('⚡ Buat & Tata Tabel Otomatis', 'setupDatabaseOtomatis') .addToUi(); } function doGet(e) { var action = (e && e.parameter) ? e.parameter.action : ''; if (action === 'setup') { setupDatabaseOtomatis(); return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'Database dibuat otomatis' })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'API SIM Presensi & SPP Bunda Ani Aktif' })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var contents = e.postData ? e.postData.contents : ''; if (!contents) return ContentService.createTextOutput(JSON.stringify({ success: false, message: 'No payload' })).setMimeType(ContentService.MimeType.JSON); var data = JSON.parse(contents); var action = data.action; if (action === 'simpanPresensi') return apiSimpanPresensi(data); if (action === 'simpanSPP') return apiSimpanSPP(data); if (action === 'tambahSiswa') return apiTambahSiswa(data); if (action === 'tambahSiswaBatch') return apiTambahSiswaBatch(data); if (action === 'syncAllData') return apiSyncAllData(data); if (action === 'ping') return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Pong!' })).setMimeType(ContentService.MimeType.JSON); return ContentService.createTextOutput(JSON.stringify({ success: false, message: 'Action tidak dikenal' })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message })).setMimeType(ContentService.MimeType.JSON); } } function apiSimpanPresensi(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Presensi') || ss.insertSheet('Presensi'); if (sheet.getLastRow() === 0) setupSheetPresensi(ss); sheet.appendRow([data.id, data.tanggal, data.waktu, data.nis, data.nama, data.kelas, data.status, data.keterangan || '-']); return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } function apiSimpanSPP(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('SPP') || ss.insertSheet('SPP'); if (sheet.getLastRow() === 0) setupSheetSPP(ss); sheet.appendRow([data.id, data.tanggal, data.nis, data.nama, data.kelas, data.bulan, data.tahun, data.nominal, data.metode, data.status || 'Lunas', data.petugas || 'Admin']); return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } function apiTambahSiswa(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Siswa') || ss.insertSheet('Siswa'); if (sheet.getLastRow() === 0) setupSheetSiswa(ss); sheet.appendRow([data.username, data.password || '123', data.nama, data.kelas, data.alamat || '-', data.sekolah_asal || '-', data.hobi || '-', data.kategori_bimbingan || 'BIMBEL', data.nama_ortu || '-', data.no_telp || '-', data.status_akun || 'Aktif']); return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } function apiTambahSiswaBatch(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Siswa') || ss.insertSheet('Siswa'); if (sheet.getLastRow() === 0) setupSheetSiswa(ss); var list = data.siswaList || []; for (var i = 0; i < list.length; i++) { var s = list[i]; sheet.appendRow([s.username, s.password || '123', s.nama, s.kelas, s.alamat || '-', s.sekolah_asal || '-', s.hobi || '-', s.kategori_bimbingan || 'BIMBEL', s.nama_ortu || '-', s.no_telp || '-', s.status_akun || 'Aktif']); } return ContentService.createTextOutput(JSON.stringify({ success: true, count: list.length })).setMimeType(ContentService.MimeType.JSON); } function apiSyncAllData(data) { setupDatabaseOtomatis(); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.siswa && data.siswa.length > 0) { var sSheet = ss.getSheetByName('Siswa'); for (var i = 0; i < data.siswa.length; i++) { var s = data.siswa[i]; sSheet.appendRow([s.username, s.password || '123', s.nama, s.kelas, s.alamat || '-', s.sekolah_asal || '-', s.hobi || '-', s.kategori_bimbingan || 'BIMBEL', s.nama_ortu || '-', s.no_telp || '-', s.status_akun || 'Aktif']); } } if (data.presensi && data.presensi.length > 0) { var pSheet = ss.getSheetByName('Presensi'); for (var j = 0; j < data.presensi.length; j++) { var p = data.presensi[j]; pSheet.appendRow([p.id, p.tanggal, p.waktu, p.nis, p.nama, p.kelas, p.status, p.keterangan || '-']); } } if (data.spp && data.spp.length > 0) { var spSheet = ss.getSheetByName('SPP'); for (var k = 0; k < data.spp.length; k++) { var sp = data.spp[k]; spSheet.appendRow([sp.id, sp.tanggal, sp.nis, sp.nama, sp.kelas, sp.bulan, sp.tahun, sp.nominal, sp.metode, sp.status || 'Lunas', sp.petugas || 'Admin']); } } return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'Sinkronisasi berhasil' })).setMimeType(ContentService.MimeType.JSON); } function setupDatabaseOtomatis() { var ss = SpreadsheetApp.getActiveSpreadsheet(); setupSheetSiswa(ss); setupSheetPresensi(ss); setupSheetSPP(ss); setupSheetUsers(ss); } function setupSheetSiswa(ss) { var s = ss.getSheetByName('Siswa') || ss.insertSheet('Siswa'); s.clear(); s.appendRow(['Username', 'Password', 'Nama Lengkap', 'Kelas', 'Alamat', 'Sekolah Asal', 'Hobi', 'Kategori Bimbingan', 'Nama Ortu', 'No Telp', 'Status Akun']); s.getRange(1, 1, 1, 11).setBackground('#4338ca').setFontColor('#ffffff').setFontWeight('bold'); s.setFrozenRows(1); } function setupSheetPresensi(ss) { var p = ss.getSheetByName('Presensi') || ss.insertSheet('Presensi'); p.clear(); p.appendRow(['ID Presensi', 'Tanggal', 'Waktu', 'Username', 'Nama Siswa', 'Kelas', 'Status Kehadiran', 'Keterangan']); p.getRange(1, 1, 1, 8).setBackground('#059669').setFontColor('#ffffff').setFontWeight('bold'); p.setFrozenRows(1); } function setupSheetSPP(ss) { var sp = ss.getSheetByName('SPP') || ss.insertSheet('SPP'); sp.clear(); sp.appendRow(['No Kuitansi', 'Tanggal Bayar', 'Username', 'Nama Siswa', 'Kelas', 'Bulan', 'Tahun', 'Nominal (Rp)', 'Metode Bayar', 'Status', 'Petugas']); sp.getRange(1, 1, 1, 11).setBackground('#d97706').setFontColor('#ffffff').setFontWeight('bold'); sp.setFrozenRows(1); } function setupSheetUsers(ss) { var u = ss.getSheetByName('Users') || ss.insertSheet('Users'); u.clear(); u.appendRow(['Nama Petugas', 'Username', 'Password', 'Role', 'Status']); u.getRange(1, 1, 1, 5).setBackground('#334155').setFontColor('#ffffff').setFontWeight('bold'); u.setFrozenRows(1); u.appendRow(['Bunda Ani (Admin)', 'admin', '123', 'admin', 'Aktif']); }
